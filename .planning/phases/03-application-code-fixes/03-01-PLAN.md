  using do
    quote do

      import Ecto

      import Ecto.Query
      import MegaPlanner.DataCase

    pid = Ecto.Adapters.SQL.Sandbox.start_owner!(MegaPlanner.Repo, shared: not tags[:async])
    on_exit(fn -> Ecto.Adapters.SQL.Sandbox.stop_owner(pid) end)

  end
end
